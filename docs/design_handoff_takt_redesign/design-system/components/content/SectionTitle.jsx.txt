import React from 'react';
export function SectionTitle({ children, dot, aside, size = 'md', style }) {
  const big = size === 'md';
  return <div style={{ display: 'flex', alignItems: 'baseline', gap: 8, margin: '0 2px', ...style }}>
    {dot ? <i style={{ width: 9, height: 9, borderRadius: '50%', background: dot, flex: 'none', alignSelf: 'center' }}></i> : null}
    <h2 style={{ margin: 0, flex: 1, fontFamily: 'var(--font-display)', fontSize: big ? 24 : 17, fontWeight: 600, letterSpacing: big ? '-.4px' : 0, color: 'var(--text)' }}>{children}</h2>
    {aside != null ? <span style={{ color: 'var(--text-muted)', fontSize: 13, fontVariantNumeric: 'tabular-nums' }}>{aside}</span> : null}
  </div>;
}
