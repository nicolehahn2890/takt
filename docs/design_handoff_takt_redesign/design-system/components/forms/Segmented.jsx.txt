import React from 'react';
export function Segmented({ options = [], value, onChange, color, style }) {
  return <div role="radiogroup" style={{ display: 'grid', gridAutoFlow: 'column', gridAutoColumns: '1fr', background: 'var(--surface-sunken)', borderRadius: 14, padding: 4, gap: 4, ...style }}>
    {options.map(o => { const v = typeof o === 'string' ? o : o.value, l = typeof o === 'string' ? o : o.label, on = v === value;
      return <button key={v} type="button" role="radio" aria-checked={on} onClick={() => onChange && onChange(v)} style={{ minHeight: 42, borderRadius: 11, border: 0, cursor: 'pointer', font: 'inherit', fontWeight: on ? 700 : 600,
        background: on ? (color || 'var(--lila)') : 'transparent', color: on ? 'var(--text-on-accent)' : 'var(--text-muted)', transition: 'background var(--dur-fast), color var(--dur-fast)' }}>{l}</button>; })}
  </div>;
}
