import React from 'react';
export function Chip({ children, selected = false, color, onClick, style }) {
  const c = color || 'var(--lila)';
  return <button type="button" onClick={onClick} aria-pressed={selected} style={{ flex: 'none', display: 'inline-flex', alignItems: 'center', gap: 7, minHeight: 40, padding: '0 14px', borderRadius: 999, border: 0, cursor: 'pointer', font: 'inherit', fontSize: 14, whiteSpace: 'nowrap',
    background: selected ? c : 'var(--surface-ghost)', color: selected ? 'var(--text-on-accent)' : 'var(--text)', fontWeight: selected ? 700 : 500, transition: 'background var(--dur-fast)', ...style }}>
    {color ? <i style={{ width: 9, height: 9, borderRadius: '50%', background: selected ? 'var(--text-on-accent)' : c, opacity: selected ? .55 : 1 }}></i> : null}{children}</button>;
}
