import React from 'react';
export function Button({ variant = 'secondary', size = 'md', block = false, theme, disabled = false, children, style, onClick, ...rest }) {
  const [press, setPress] = React.useState(false);
  const sizes = { sm: { minHeight: 44, padding: '0 16px', fontSize: 15, borderRadius: 12 }, md: { minHeight: 'var(--button-h)', padding: '0 20px', fontSize: 16 }, lg: { minHeight: 'var(--button-h-lg)', padding: '0 24px', fontSize: 17, borderRadius: 16 } };
  const t = theme || 'var(--accent)';
  const variants = {
    primary: { background: t, color: 'var(--text-on-accent)' },
    secondary: { background: 'var(--surface-ghost)', color: 'var(--text)', fontWeight: 600 },
    ghost: { background: 'transparent', color: 'var(--text)', fontWeight: 600 },
    danger: { background: 'var(--danger-surface)', color: 'var(--danger)' },
    pill: { background: 'var(--lila)', color: 'var(--text-on-accent)', borderRadius: 999 },
    link: { background: 'transparent', color: 'var(--text-muted)', fontWeight: 500, fontSize: 13, padding: '0 4px', minHeight: 34, textDecoration: 'underline', textUnderlineOffset: 3, textDecorationColor: 'var(--line-strong)' }
  };
  const s = {
    font: 'inherit', fontWeight: 700, border: 0, cursor: disabled ? 'default' : 'pointer', display: block ? 'flex' : 'inline-flex', alignItems: 'center', justifyContent: 'center', gap: 8,
    borderRadius: 'var(--radius-control)', width: block ? '100%' : undefined, touchAction: 'manipulation', whiteSpace: 'nowrap',
    transition: 'transform var(--dur-fast) var(--ease-standard), filter var(--dur-fast)', transform: press && !disabled ? 'scale(var(--press-scale))' : 'none', opacity: disabled ? .4 : 1,
    ...sizes[size], ...variants[variant], ...style
  };
  return <button type="button" disabled={disabled} onClick={onClick} onPointerDown={() => setPress(true)} onPointerUp={() => setPress(false)} onPointerLeave={() => setPress(false)} style={s} {...rest}>{children}</button>;
}
