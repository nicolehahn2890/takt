const exById = id => (window.Takt.EXERCISES || []).find(e => e.id === id) || { id, name: id, alias: [], tags: [] };
const fmt = s => { s = Math.max(0, Math.floor(s || 0)); return Math.floor(s / 60) + ':' + ('0' + (s % 60)).slice(-2); };
function buildSteps(w, k = 1) {
  const steps = [{ kind: 'prep', duration: 5 }];
  w.blocks.forEach(b => { for (let r = 1; r <= b.repeat; r++) b.items.forEach(it => {
    const ex = exById(it.exId);
    steps.push({ kind: 'work', exId: it.exId, name: ex.name, duration: Math.max(10, Math.round(it.duration * k / 5) * 5), round: r, rounds: b.repeat });
    if (w.rest) steps.push({ kind: 'rest', duration: w.rest });
  }); });
  if (steps[steps.length - 1].kind === 'rest') steps.pop();
  return steps;
}
const secsOf = w => buildSteps(w).slice(1).reduce((a, s) => a + s.duration, 0);
const countOf = w => buildSteps(w).filter(s => s.kind === 'work').length;
const mins = s => Math.max(1, Math.round(s / 60)) + ' Min.';
function cardIds(w, n) { const out = []; w.blocks.slice().sort((a, b) => b.repeat - a.repeat).forEach(b => b.items.forEach(i => { if (out.length < n && out.indexOf(i.exId) < 0) out.push(i.exId); })); return out; }
